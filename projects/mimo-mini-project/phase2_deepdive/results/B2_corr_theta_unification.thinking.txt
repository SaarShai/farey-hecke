from fractions import Fraction
import numpy as np
from scipy import stats

def farey_sequence(N):
    """Generate Farey sequence of order N"""
    seq = []
    a, b, c, d = 0, 1, 1, N
    seq.append(Fraction(a, b))
    while c <= N:
        k = (N + b) // d
        a, b, c, d = c, d, k * c - a, k * d - b
        seq.append(Fraction(a, b))
    return seq

def compute_gaps(N):
    """Compute gaps in Farey sequence of order N"""
    seq = farey_sequence(N)
    gaps = [float(seq[i+1] - seq[i]) for i in range(len(seq) - 1)]
    return np.array(gaps)

# Compute for several N values
for N in [50, 100, 200, 500, 1000]:
    gaps = compute_gaps(N)
    
    # Lag-1 Pearson correlation
    d1 = gaps[:-1]
    d2 = gaps[1:]
    corr = np.corrcoef(d1, d2)[0, 1]
    
    # Extremal index (runs estimator)
    threshold = np.percentile(gaps, 95)
    exceedances = gaps > threshold
    n_exceed = np.sum(exceedances)
    n_clusters = np.sum(exceedances[:-1] & ~exceedances[1:]) + exceedances[0]  # number of cluster starts
    if n_exceed > 0:
        theta_runs = n_clusters / n_exceed
    else:
        theta_runs = float('nan')
    
    print(f"N={N}: corr={corr:.4f}, theta_runs={theta_runs:.4f}, n_gaps={len(gaps)}")

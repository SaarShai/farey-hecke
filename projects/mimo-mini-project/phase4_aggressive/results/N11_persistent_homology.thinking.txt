import numpy as np
from scipy.spatial.distance import pdist, squareform
from ripser import ripser
from persim import plot_diagrams
import matplotlib.pyplot as plt
from math import gcd

def farey_points(N):
    """
    Generate points on the unit circle corresponding to Farey fractions of order N.
    Returns list of (x,y) coordinates.
    """
    points = []
    for q in range(1, N+1):
        for p in range(q):
            if gcd(p, q) == 1:
                angle = 2 * np.pi * p / q
                x = np.cos(angle)
                y = np.sin(angle)
                points.append((x, y))
    return np.array(points)

def main():
    N = 100
    print(f"Generating Farey points for N={N}...")
    pts = farey_points(N)
    print(f"Number of points: {len(pts)}")
    
    # Compute pairwise Euclidean distances
    print("Computing pairwise distances...")
    dists = pdist(pts, metric='euclidean')
    dist_matrix = squareform(dists)
    
    # Compute persistence with ripser (max dimension 1)
    print("Computing persistent homology...")
    diagrams = ripser(dist_matrix, maxdim=1, distance_matrix=True)
    
    # Extract persistence for dimension 1
    pers_1 = diagrams['dgms'][1]  # array of (birth, death)
    
    # Compute lifetime
    lifetimes = pers_1[:, 1] - pers_1[:, 0]
    
    # Sort by lifetime descending
    idx = np.argsort(lifetimes)[::-1]
    pers_1_sorted = pers_1[idx]
    lifetimes_sorted = lifetimes[idx]
    
    print("\nTop 5 persistent H1 features (birth, death, lifetime):")
    for i in range(min(5, len(pers_1_sorted))):
        b, d = pers_1_sorted[i]
        life = lifetimes_sorted[i]
        print(f"Feature {i+1}: birth={b:.6f}, death={d:.6f}, lifetime={life:.6f}, ratio death/birth={d/b if b>0 else float('inf'):.6f}")
    
    # Check the second feature (index 1)
    if len(pers_1_sorted) >= 2:
        b2, d2 = pers_1_sorted[1]
        ratio2 = d2 / b2 if b2 > 0 else float('inf')
        print(f"\nSecond persistent cycle: birth={b2:.6f}, death={d2:.6f}, ratio death/birth={ratio2:.6f}")
        if abs(ratio2 - 3) < 0.5:
            print("=> The ratio is approximately 3, supporting the claim.")
        else:
            print(f"=> The ratio is {ratio2:.4f}, which is not approximately 3.")
    else:
        print("\nNo second persistent cycle found.")
    
    # Plot persistence diagram
    plt.figure(figsize=(8, 8))
    plot_diagrams(diagrams['dgms'], show=False)
    plt.title(f"Persistence Diagram (H1) for Farey N={N}")
    plt.savefig(f"farey_persistence_N{N}.png")
    print(f"\nPersistence diagram saved to farey_persistence_N{N}.png")
    
if __name__ == "__main__":
    main()
